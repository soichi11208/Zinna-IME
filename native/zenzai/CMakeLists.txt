cmake_minimum_required(VERSION 3.22)
project(zinna_zenzai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
if(NOT ZINNA_LLAMA_SOURCE)
    set(ZINNA_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/zenzai-llama")
endif()
add_subdirectory("${ZINNA_LLAMA_SOURCE}" llama)
add_library(zenzai_core STATIC engine.cpp)
target_link_libraries(zenzai_core PUBLIC llama)
if(ANDROID)
    add_library(zinna_zenzai SHARED jni.cpp)
    target_link_libraries(zinna_zenzai PRIVATE zenzai_core log)
    target_link_options(zinna_zenzai PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(zenzai-check check.cpp)
    target_link_libraries(zenzai-check PRIVATE zenzai_core)
endif()
